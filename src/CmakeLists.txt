file(GLOB_RECURSE ASTRA_SOURCES
    "*.cpp"
    "*.cxx"
    "*.c"
)
file(GLOB_RECURSE ASTRA_HEADERS
    "../include/astra/*.h"
    "../include/astra/*.hpp" 
    "../include/astra/*.inl"
)
add_library(astra_static STATIC ${ASTRA_SOURCES})

set_target_properties(astra_static PROPERTIES
    OUTPUT_NAME "astra"
    PUBLIC_HEADER "${ASTRA_HEADERS}"
)
target_include_directories(astra_static PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/../include
)
target_compile_options(astra_static PRIVATE
    $<$<CXX_COMPILER_ID:MSVC>:/Zi>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-g>
)
